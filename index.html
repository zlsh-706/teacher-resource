detailModal.style.display = 'none'; // [關鍵修正] 強制寫入 inline style } // 2.3 恢復 Body 滾動 (移除所有可能的 inline styles) const body = clone.querySelector('body'); if (body) { body.removeAttribute('style'); // 直接移除 style 屬性，確保 overflow: hidden 被清空 body.classList.remove('overflow-hidden'); } // 3. 取得 "淨化後" 的 HTML 字串 let html = clone.outerHTML; // 4. 準備要寫入的 JSON 字串 (將當前 LocalStorage 的最新資料轉為字串) const jsonResources = JSON.stringify(state.resources, null, 4); // 格式化縮排 const jsonCategories = JSON.stringify(state.categories, null, 4);
